#define W_ENUM_VALUE_TO_STRING(name) \
  case name: return W_STRINGIZE(name);
#define W_ENUM_TO_STRING(...) \
  const char* ToString(WUInt32 value) \
  { \
    switch (value) \
    { \
      W_EXPAND_ARGS(W_ENUM_VALUE_TO_STRING, ##__VA_ARGS__) \
      default: return nullptr; \
    } \
  }
#define W_DECLARE_FLAGS_OPERATORS(FlagsType) \
  inline WBitflags<FlagsType> operator|(FlagsType::Enum lhs, FlagsType::Enum rhs)    \
  {    \
    return (WBitflags<FlagsType>(lhs) | WBitflags<FlagsType>(rhs));    \
  } \
  \
  inline WBitflags<FlagsType> operator&(FlagsType::Enum lhs, FlagsType::Enum rhs)    \
  {    \
    return (WBitflags<FlagsType>(lhs) & WBitflags<FlagsType>(rhs));    \
  }
#define W_DECLARE_FLAGS(InternalStorageType, BitflagsTypeName, ...)    \
struct BitflagsTypeName    \
  {    \
    static const WUInt32 Count = W_VA_NUM_ARGS(__VA_ARGS__);    \
    typedef InternalStorageType StorageType; \
    enum Enum    \
    {    \
      W_EXPAND_ARGS_WITH_INDEX(W_DECLARE_FLAGS_ENUM, ##__VA_ARGS__)    \
    };    \
    struct Bits    \
    {    \
      W_EXPAND_ARGS(W_DECLARE_FLAGS_BITS, ##__VA_ARGS__)    \
    };    \
    W_ENUM_TO_STRING(__VA_ARGS__) \
  };    \
  W_DECLARE_FLAGS_OPERATORS(BitflagsTypeName)
#define W_DECLARE_FLAGS_ENUM(name, n)    name = W_BIT(n),
#define W_DECLARE_FLAGS_BITS(name)       StorageType name : 1;
